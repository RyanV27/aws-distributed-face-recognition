# Step 1: Build the Docker image locally
docker build --no-cache -t project-2-part-1-lambda:latest .
docker build -t project-2-part-1-lambda:latest .


# Step 2: Create an ECR repository
Go to AWS Management Console → ECR → Repositories → Create repository.
Give it a name, e.g., project-2.
Set visibility: Private (default).

Repository URI: <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com/project-2


# Step 3: Authenticate Docker to ECR
Prerequisite:
Adding 'AmazonEC2ContainerRegistryFullAccess' permission policy to cse546-Developer

aws ecr get-login-password --region us-east-1 | docker login \
    --username AWS \
    --password-stdin <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com
Terminal Output: WARNING! Your credentials are stored unencrypted in '/home/ryan/.docker/config.json'.


# Step 4: Tag and push the Docker image
# Tag the image for ECR
docker tag project-2-part-1-lambda:latest <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com/project-2-part-1:latest

docker images
docker rmi

# Push the image to ECR
docker push <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com/project-2-part-1:latest


# Step 5: Create Lambda functions using the same container image
Option A: AWS Console
Go to AWS Lambda → Create function → Container image.
Give function name: e.g., face-detection.
Select the container image from ECR: face-recognition-lambda:latest.
In the Image URI configuration, set Handler override:
    fd_lambda.handler
Configure memory, timeout, etc.
Repeat for face-recognition function, with Handler override:
    fr_lambda.handler

# ------------------------------------------------------------------ #
Updating lambda through cli:
aws lambda update-function-code \
  --function-name face-detection \
  --image-uri <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com/project-2-part-1:latest

aws lambda update-function-code \
  --function-name face-recognition \
  --image-uri <AWS_ACCOUNT_ID>.dkr.ecr.us-east-1.amazonaws.com/project-2-part-1:latest
